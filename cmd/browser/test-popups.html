Usage Examples 

// Basic usage from any component:
const popupManager = this.closest('popup-manager')
const popup = document.createElement('view-popup')
popup.setAttribute('title', 'My Popup')
popup.innerHTML = '<p>Hello World!</p>'
popupManager.appendChild(popup)

// Using the convenience API:
popupManager.showPopup({
    title: 'My Popup',
    content: '<p>Hello World!</p>',
    size: 'medium'
})
